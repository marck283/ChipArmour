82static inline void set_mpu_region(mpu_region_t region) {
83 uint32_t rasr_attr;
84 switch (region.access_type) {
85 case DEVICE: {
86 rasr_attr = ARM_MPU_ACCESS_DEVICE(region.shareable);
87 break;
88 }
89 case ORDERED: {
90 rasr_attr = ARM_MPU_ACCESS_ORDERED;
91 break;
92 }
93 case NORMAL:
94 default: {
95 rasr_attr = ARM_MPU_ACCESS_NORMAL(region.outer_cp, region.inner_cp, region.shareable);
96 break;
97 }
98 }

127static inline uint8_t init_mpu(mpu_region_t regions[MPU_REGIONS_NUM]) {
128 if (regions == NULL) { // Check for NULL pointer input
129 return ENOPOINTER; // Handle NULL pointer input gracefully, though this should not happen in practice
130 }
131
132 for(uint8_t i = 0; i < MPU_REGIONS_NUM; i++) {
133 set_mpu_region(regions[i]); // This is an observable side-effect, so the compiler cannot optimize this for loop away
134 }
135
136 ARM_MPU_Enable(MPU_CTRL_PRIVDEFENA_Msk);
137
138 return SUCCESS; // Return success
139}
