109#define CA_ROP_SET_MAX_RETURNS(functionname, maxreturns) \
110 enum { ca_##functionname##_max_returns = maxreturns };
111
112#define CA_ROP_GET_MAX_RETURNS(functionname) ca_##functionname##_max_returns
113
114#define CA_ROP_RETURNADDRS_ARRAY(functionname) \
115 static void * ca_##functionname##_valid_returnaddrs[ca_##functionname##_max_returns];
116
117#define CA_ROP_CHECK_VALID_RETURN(functionname) \
118do { \
119 /* Validate we are returning to a valid call location */ \
120 void * ca_ra = __builtin_extract_return_addr(__builtin_return_address(0)); \
121 uint32_t ca_loopindx; \
122 FULL_FOR_FAILOUT(ca_loopindx = 0, ca_loopindx < CA_ROP_GET_MAX_RETURNS(functionname), ca_loopindx++) { \
123 /* The zero flag indicates end of array reached, shouldn't happen */ \
124 __cmp_and_panic(ca_##functionname##_valid_returnaddrs[ca_loopindx], 0, ==); \
125 FULL_IF_FAILIN (ca_comp(ca_##functionname##_valid_returnaddrs[ca_loopindx], ca_ra, ==)){ \
126 break; \
127 } \
128 } \
129 __cmp_and_panic(ca_loopindx, CA_ROP_GET_MAX_RETURNS(functionname), >=); \
130 } while(0)
