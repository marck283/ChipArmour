78static inline void set_mpu_region(mpu_region_t region) {
79 uint32_t rasr_attr;
80 switch (region.access_type) {
81 case DEVICE: {
82 rasr_attr = ARM_MPU_ACCESS_DEVICE(region.shareable);
83 break;
84 }
85 case ORDERED: {
86 rasr_attr = ARM_MPU_ACCESS_ORDERED;
87 break;
88 }
89 case NORMAL:
90 default: {
91 rasr_attr = ARM_MPU_ACCESS_NORMAL(region.outer_cp, region.inner_cp, region.shareable);
92 break;
93 }
94 }

121static inline uint8_t init_mpu(mpu_region_t regions[MPU_REGIONS_NUM]) {
122 if (regions == NULL) {
123 return ENOPOINTER; // Handle null pointer input gracefully, though this should not happen in practice
124 }
125
126 for(uint8_t i = 0; i < MPU_REGIONS_NUM; i++) {
127 set_mpu_region(regions[i]); // This is an observable side-effect, so the compiler cannot optimize this for loop away
128 }
129
130 ARM_MPU_Enable(MPU_CTRL_PRIVDEFENA_Msk);
131
132 return SUCCESS; // Return success
133}
