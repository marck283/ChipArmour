72static inline void set_mpu_region(mpu_region_t region) {
73 uint32_t rasr_attr;
74 switch (region.access_type) {
75 case DEVICE: {
76 rasr_attr = ARM_MPU_ACCESS_DEVICE(region.shareable);
77 break;
78 }
79 case ORDERED: {
80 rasr_attr = ARM_MPU_ACCESS_ORDERED;
81 break;
82 }
83 case NORMAL:
84 default: {
85 rasr_attr = ARM_MPU_ACCESS_NORMAL(region.outer_cp, region.inner_cp, region.shareable);
86 break;
87 }
88 }

93 region.region_size));
94}

116static inline uint8_t init_mpu(mpu_region_t regions[MPU_REGIONS_NUM]) {
117 if (regions == NULL) {
118 return ENOPOINTER; // Handle null pointer input gracefully, though this should not happen in practice
119 }
120
121 for(uint8_t i = 0; i < MPU_REGIONS_NUM; i++) {
122 set_mpu_region(regions[i]); // This is an observable side-effect, so the compiler cannot optimize this for loop away
123 }
124
125 ARM_MPU_Enable(MPU_CTRL_PRIVDEFENA_Msk);
126
127 return SUCCESS; // Return success
128}
